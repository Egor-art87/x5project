# --- Web-фреймворк ---
fastapi             # сам фреймворк: роуты, валидация, OpenAPI-доки
uvicorn[standard]   # ASGI-сервер, на котором запускается FastAPI

# --- Работа с БД ---
sqlalchemy[asyncio] # ORM: описываем таблицы как Python-классы
asyncpg             # асинхронный драйвер PostgreSQL (для прода)
aiosqlite           # асинхронный драйвер SQLite (для локальной разработки)
alembic             # миграции БД (версионирование схемы)

# --- Валидация и настройки ---
pydantic            # схемы данных (что приходит/уходит по API)
pydantic-settings   # читает настройки из .env файла

# --- Безопасность ---
passlib[bcrypt]     # хеширование паролей
pyjwt               # выпуск/проверка JWT-токенов
python-multipart    # парсинг form-data (нужно FastAPI для логина)

# --- HTTP-клиент (для GigaChat позже) ---
httpx

# --- Кэш/очередь (на будущее) ---
redis
